<body>
  <h2>Чат для своих</h2>
  <div id="messages"></div>
  <input type="text" id="text" placeholder="Сообщение...">
  <button onclick="send()">Отправить</button>

  <script>
    const messages = document.getElementById('messages');
    const textInput = document.getElementById('text');

    function loadMessages() {
      const stored = localStorage.getItem('myChatMessages') || '[]';
      const arr = JSON.parse(stored);
      messages.innerHTML = '';
      arr.forEach(m => {
        const div = document.createElement('div');
        div.className = 'msg';
        div.textContent = m;
        messages.appendChild(div);
      });
      messages.scrollTop = messages.scrollHeight;
    }

    function send() {
      const val = textInput.value.trim();
      if (!val) return;
      const now = new Date().toLocaleTimeString();
      const msg = `[${now}] Ты: ${val}`;

      const arr = JSON.parse(localStorage.getItem('myChatMessages') || '[]');
      arr.push(msg);
      if (arr.length > 100) arr.shift();
      localStorage.setItem('myChatMessages', JSON.stringify(arr));

      loadMessages();
      textInput.value = '';
    }

    loadMessages();
  </script>
